import datetime

fridays = []
for year in range(2026, 2031):
    for month in range(1, 13):
        d = datetime.date(year, month, 13)
        if d.weekday() == 4: # Monday=0, Friday=4
            fridays.append(d)

for d in fridays:
    print(d.isoformat(), d.strftime("%A"))

print(f"\nTotal: {len(fridays)}")
